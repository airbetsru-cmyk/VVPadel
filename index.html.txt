    <!-- Шапка приложения -->
    <div class="flex items-center justify-between bg-white p-4 rounded-2xl shadow-sm border border-gray-100">
      <div>
        <h1 class="text-lg font-bold text-emerald-700">Padel Moldova 🎾</h1>
        <p id="user-greeting" class="text-xs text-gray-500">Авторизация...</p>

    <!-- Главный контейнер (динамически меняется через JS) -->

      <div class="flex justify-center items-center py-12 text-gray-400">Загрузка данных...</div>
    </div>
  </div>

  <script>
    // 1. Инициализация Telegram SDK
    const tg = window.Telegram.WebApp;
    tg.ready();
    tg.expand();

    // 2. Инициализация Supabase (ВСТАВЬТЕ ВАШИ ДАННЫЕ НИЖЕ)
    const SUPABASE_URL = 'https://ppdxzmcycwhtuhebaduh.supabase.co';
    const SUPABASE_KEY = 'sb_publishable_KRbMwTasNQpUqQ7gKihOxg_KKQiUbm-';
    const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

    // Данные пользователя Telegram
    const tgUser = tg.initDataUnsafe?.user || {
      id: 999000111,
      first_name: 'Тестовый',
      last_name: 'Игрок',
      username: 'testplayer'
    };

    // Определение контекста запуска (создание игры или переход по ссылке)
    const startParam = tg.initDataUnsafe?.start_param || new URLSearchParams(window.location.search).get('startapp');

    // Старт приложения
    async function initApp() {
      await syncUserProfile();
      document.getElementById('user-greeting').innerText = `${tgUser.first_name} ${tgUser.last_name || ''}`;

      if (startParam) {
        renderMatchDetails(startParam);
      } else {
        renderCreateMatchForm();
      }
    }

    // Синхронизация профиля с таблицей profiles
    async function syncUserProfile() {
      await supabase.from('profiles').upsert({
        id: tgUser.id,
        first_name: tgUser.first_name,
        last_name: tgUser.last_name || null,
        username: tgUser.username || null,
        updated_at: new Date()
      });
    }

    // Форма создания игры
    function renderCreateMatchForm() {
      const container = document.getElementById('main-content');
      container.innerHTML = `
        <h2 class="text-base font-bold mb-4 text-gray-800">Создать новую игру</h2>
        <form id="create-form" class="space-y-3" onsubmit="handleCreateMatch(event)">
          <div>
            <label class="block text-xs font-medium text-gray-600 mb-1">Локация / Корт</label>
            <input type="text" id="custom_location" required value="Padel Club Chisinau" placeholder="Например: Арена Кишинёв" class="w-full p-2.5 bg-gray-50 border border-gray-200 rounded-xl text-sm focus:outline-none focus:border-emerald-500">
          </div>

          <div class="grid grid-cols-2 gap-2">
            <div>
              <label class="block text-xs font-medium text-gray-600 mb-1">Количество мест</label>
              <select id="max_players" class="w-full p-2.5 bg-gray-50 border border-gray-200 rounded-xl text-sm font-medium">
                <option value="4">4 игрока (2х2)</option>
                <option value="6">6 игроков (Americano)</option>
                <option value="8">8 игроков (2 корта)</option>
              </select>
            </div>
            <div>
              <label class="block text-xs font-medium text-gray-600 mb-1">Формат</label>
              <select id="match_type" class="w-full p-2.5 bg-gray-50 border border-gray-200 rounded-xl text-sm font-medium">
                <option value="friendly">Дружеский</option>
                <option value="competitive">Рейтинговый</option>
                <option value="americano">Americano</option>
              </select>
            </div>
          </div>

          <div class="grid grid-cols-2 gap-2">
            <div>
              <label class="block text-xs font-medium text-gray-600 mb-1">Дата и время</label>
              <input type="datetime-local" id="scheduled_at" required class="w-full p-2.5 bg-gray-50 border border-gray-200 rounded-xl text-sm">
            </div>
            <div>
              <label class="block text-xs font-medium text-gray-600 mb-1">Цена с игрока (MDL)</label>
              <input type="number" id="price_per_player" value="150" min="0" step="10" class="w-full p-2.5 bg-gray-50 border border-gray-200 rounded-xl text-sm">
            </div>
          </div>

          <div>
            <label class="block text-xs font-medium text-gray-600 mb-1">Заметки / Пожелания</label>
            <input type="text" id="notes" placeholder="Мячи свои, уровень 2.5+" class="w-full p-2.5 bg-gray-50 border border-gray-200 rounded-xl text-sm">
          </div>

          <button type="submit" id="submit-btn" class="w-full py-3 bg-emerald-600 text-white font-bold rounded-xl shadow-md hover:bg-emerald-700 active:scale-95 transition-all text-sm mt-2">
            Сформировать ссылку на игру
          </button>
        </form>
      `;

      // Установка значения даты по умолчанию (+3 часа от текущего)
      const defaultTime = new Date(Date.now() + 3 * 3600 * 1000).toISOString().slice(0, 16);
      document.getElementById('scheduled_at').value = defaultTime;
    }

    // Обработка создания игры
    async function handleCreateMatch(e) {
      e.preventDefault();
      const btn = document.getElementById('submit-btn');
      btn.disabled = true;
      btn.innerText = 'Создание...';

      const matchData = {
        creator_id: tgUser.id,
        custom_location: document.getElementById('custom_location').value,
        max_players: parseInt(document.getElementById('max_players').value),
        match_type: document.getElementById('match_type').value,
        scheduled_at: new Date(document.getElementById('scheduled_at').value).toISOString(),
        price_per_player: parseFloat(document.getElementById('price_per_player').value) || 0,
        notes: document.getElementById('notes').value
      };

      // 1. Создаем запись в matches
      const { data: match, error } = await supabase.from('matches').insert([matchData]).select().single();

      if (error) {
        alert('Ошибка при создании игры: ' + error.message);
        btn.disabled = false;
        btn.innerText = 'Сформировать ссылку';
        return;
      }

      // 2. Добавляем создателя в слот №1
      await supabase.from('match_players').insert([{
        match_id: match.id,
        user_id: tgUser.id,
        slot_number: 1
      }]);

      renderMatchDetails(match.id);
    }

    // Отображение деталей игры и списка участников
    async function renderMatchDetails(matchId) {
      const container = document.getElementById('main-content');
      container.innerHTML = `<div class="text-center py-8 text-gray-400">Загрузка карточки матча...</div>`;

      // Запрашиваем данные из нашего SQL-представления view_active_matches
      const { data: match, error } = await supabase
        .from('view_active_matches')
        .select('*')
        .eq('match_id', matchId)
        .single();

      if (error || !match) {
        container.innerHTML = `<div class="text-red-500 font-medium text-center">Игра не найдена или была удалена.</div>`;
        return;
      }

      const players = match.players || [];
      const isJoined = players.some(p => p.user_id === tgUser.id);
      const isFull = players.length >= match.max_players;

      const dateStr = new Date(match.scheduled_at).toLocaleString('ru-RU', {
        day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit'
      });

      // Формирование слотов
      let slotsHtml = '';
      for (let i = 1; i <= match.max_players; i++) {
        const playerInSlot = players.find(p => p.slot_number === i || (!p.slot_number && players.indexOf(p) === i - 1));
        if (playerInSlot) {
          slotsHtml += `
            <div class="flex items-center justify-between p-2.5 bg-emerald-50 border border-emerald-100 rounded-xl">
              <span class="text-xs font-semibold text-emerald-900">Слот #${i}: ${playerInSlot.first_name}</span>
              <span class="text-[10px] bg-emerald-200 text-emerald-800 px-2 py-0.5 rounded-full">Занят</span>
            </div>`;
        } else {
          slotsHtml += `
            <div class="flex items-center justify-between p-2.5 bg-gray-50 border border-dashed border-gray-300 rounded-xl text-gray-400">
              <span class="text-xs">Слот #${i}: Свободно</span>
            </div>`;
        }
      }

      const shareBotUsername = 'ВАШ_ЮЗЕРНЕЙМ_БОТА'; // Замените на имя вашего бота (без @)
      const shareUrl = `https://t.me/${shareBotUsername}/app?startapp=${match.match_id}`;

      container.innerHTML = `
        <div class="space-y-4">
          <div>
            <div class="flex justify-between items-start">
              <span class="px-2 py-0.5 bg-blue-100 text-blue-800 text-[11px] font-bold uppercase tracking-wider rounded">
                ${match.match_type}
              </span>
              <span class="text-xs font-bold text-emerald-700">${match.price_per_player} MDL / чел</span>
            </div>
            <h2 class="text-lg font-bold mt-1 text-gray-900">${match.location_name}</h2>
            <p class="text-xs text-gray-500">📅 ${dateStr}</p>
            ${match.notes ? `<p class="text-xs italic text-gray-600 mt-1">"${match.notes}"</p>` : ''}
          </div>

          <div class="border-t border-gray-100 pt-3">
            <h3 class="text-xs font-bold text-gray-700 mb-2">Участники (${players.length}/${match.max_players})</h3>
            <div class="space-y-1.5">
              ${slotsHtml}
            </div>
          </div>

          <div class="space-y-2 pt-2">
            ${!isJoined && !isFull ? `
              <button onclick="joinMatch('${match.match_id}',${players.length + 1})" class="w-full py-3 bg-emerald-600 text-white font-bold rounded-xl shadow-md hover:bg-emerald-700 active:scale-95 transition-all text-sm">
                Занять слот
              </button>
            ` : ''}

            ${isJoined ? `
              <button onclick="leaveMatch('${match.match_id}')" class="w-full py-2.5 bg-red-50 text-red-600 font-semibold rounded-xl text-xs hover:bg-red-100">
                Покинуть игру
              </button>
            ` : ''}

            <button onclick="copyShareLink('${shareUrl}')" class="w-full py-2.5 bg-gray-100 text-gray-800 font-semibold rounded-xl text-xs hover:bg-gray-200">
              📋 Скопировать ссылку на игру
            </button>
          </div>
        </div>
      `;
    }

    // Присоединиться к игре
    async function joinMatch(matchId, nextSlot) {
      const { error } = await supabase.from('match_players').insert([{
        match_id: matchId,
        user_id: tgUser.id,
        slot_number: nextSlot
      }]);

      if (error) {
        alert('Не удалось занять место: ' + error.message);
      } else {
        renderMatchDetails(matchId);
      }
    }

    // Выйти из игры
    async function leaveMatch(matchId) {
      const { error } = await supabase.from('match_players').delete()
        .eq('match_id', matchId)
        .eq('user_id', tgUser.id);

      if (error) {
        alert('Ошибка: ' + error.message);
      } else {
        renderMatchDetails(matchId);
      }
    }

    // Поделиться ссылкой
    function copyShareLink(url) {
      if (navigator.clipboard) {
        navigator.clipboard.writeText(url);
        tg.showAlert('Ссылка на бронирование скопирована! Отправьте её игрокам.');
      } else {
        prompt('Скопируйте ссылку вручную:', url);
      }
    }

    // Запуск приложения
    initApp();
  </script>
